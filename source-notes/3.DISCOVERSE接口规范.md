# DISCOVERSE 接口设计规范

## 1. 概述

本文档定义了 DISCOVERSE 系统的核心接口规范，旨在：
- 提供清晰的模块边界和职责划分
- 支持不同 IK 求解器的替换
- 支持仿真环境和真实机器人的切换
- 便于单元测试和集成测试

## 2. 核心抽象接口

### 2.1 IKSolver 接口

**职责**: 求解逆运动学问题

```python
from abc import ABC, abstractmethod
from typing import Tuple, Optional
import numpy as np

class SE3:
    """SE(3) 刚体变换"""
    def __init__(self, position: np.ndarray, rotation: np.ndarray):
        """
        Args:
            position: [x, y, z]
            rotation: 3×3 旋转矩阵 或 四元数 [w,x,y,z]
        """
        self.position = position
        self.rotation = rotation
    
    def to_matrix(self) -> np.ndarray:
        """转换为 4×4 齐次变换矩阵"""
        pass
    
    def inverse(self) -> 'SE3':
        """求逆变换"""
        pass
    
    @classmethod
    def from_matrix(cls, matrix: np.ndarray) -> 'SE3':
        """从 4×4 矩阵构造"""
        pass


class IKSolver(ABC):
    """逆运动学求解器抽象接口"""
    
    @abstractmethod
    def solve(self,
              target_pose: SE3,
              current_q: np.ndarray,
              dt: float,
              max_iterations: int = 10,
              position_threshold: float = 1e-3,
              orientation_threshold: float = 1e-3
              ) -> Tuple[np.ndarray, bool]:
        """
        求解 IK 问题
        
        Args:
            target_pose: 目标末端位姿
            current_q: 当前关节角度 (n,)
            dt: 时间步长 (s)
            max_iterations: 最大迭代次数
            position_threshold: 位置收敛阈值 (m)
            orientation_threshold: 姿态收敛阈值 (rad)
        
        Returns:
            q_target: 目标关节角度 (n,)
            converged: 是否收敛
        """
        pass
    
    @abstractmethod
    def compute_jacobian(self, q: np.ndarray) -> np.ndarray:
        """
        计算雅可比矩阵
        
        Args:
            q: 关节角度 (n,)
        
        Returns:
            J: 雅可比矩阵 (6×n)
        """
        pass
    
    @abstractmethod
    def forward_kinematics(self, q: np.ndarray) -> SE3:
        """
        正向运动学
        
        Args:
            q: 关节角度 (n,)
        
        Returns:
            pose: 末端位姿
        """
        pass
```

**实现示例**: Mink IK Solver

```python
class MinkIKSolver(IKSolver):
    """基于 mink 库的 IK 求解器"""
    
    def __init__(self, 
                 mj_model,
                 end_effector_frame: str = "endpoint",
                 frame_type: str = "site",
                 position_cost: float = 100.0,
                 orientation_cost: float = 10.0,
                 posture_cost: float = 1e-2):
        self.model = mj_model
        self.configuration = mink.Configuration(mj_model)
        
        # 设置任务
        self.end_effector_task = mink.FrameTask(
            frame_name=end_effector_frame,
            frame_type=frame_type,
            position_cost=position_cost,
            orientation_cost=orientation_cost,
        )
        self.posture_task = mink.PostureTask(
            model=mj_model,
            cost=posture_cost
        )
        self.tasks = [self.end_effector_task, self.posture_task]
    
    def solve(self, target_pose, current_q, dt, **kwargs):
        max_iters = kwargs.get('max_iterations', 10)
        pos_thresh = kwargs.get('position_threshold', 1e-3)
        ori_thresh = kwargs.get('orientation_threshold', 1e-3)
        
        # 更新配置
        self.configuration.update(current_q)
        
        # 设置目标
        mink_target = self._se3_to_mink(target_pose)
        self.end_effector_task.set_target(mink_target)
        
        # 迭代求解
        for i in range(max_iters):
            vel = mink.solve_ik(self.configuration, self.tasks, dt, "quadprog")
            self.configuration.integrate_inplace(vel, dt)
            
            err = self.end_effector_task.compute_error(self.configuration)
            pos_ok = np.linalg.norm(err[:3]) <= pos_thresh
            ori_ok = np.linalg.norm(err[3:]) <= ori_thresh
            
            if pos_ok and ori_ok:
                return self.configuration.q, True
        
        return self.configuration.q, False
```

### 2.2 RobotController 接口

**职责**: 机器人控制命令发送和状态读取

```python
from dataclasses import dataclass
from enum import Enum

class ControlMode(Enum):
    """控制模式"""
    POSITION = "position"
    VELOCITY = "velocity"
    TORQUE = "torque"
    IMPEDANCE = "impedance"


@dataclass
class JointState:
    """关节状态"""
    position: np.ndarray  # 位置 (rad or m)
    velocity: np.ndarray  # 速度 (rad/s or m/s)
    effort: np.ndarray    # 力矩/力 (N·m or N)
    timestamp: float      # 时间戳 (s)


class RobotController(ABC):
    """机器人控制器抽象接口"""
    
    @abstractmethod
    def connect(self) -> bool:
        """
        连接到机器人
        
        Returns:
            success: 是否连接成功
        """
        pass
    
    @abstractmethod
    def disconnect(self) -> None:
        """断开连接"""
        pass
    
    @abstractmethod
    def get_joint_state(self) -> JointState:
        """
        获取关节状态
        
        Returns:
            state: 关节状态
        """
        pass
    
    @abstractmethod
    def set_joint_command(self, 
                         mode: ControlMode,
                         command: np.ndarray) -> bool:
        """
        发送关节命令
        
        Args:
            mode: 控制模式
            command: 命令值 (n,)
        
        Returns:
            success: 是否发送成功
        """
        pass
    
    @abstractmethod
    def get_end_effector_pose(self) -> SE3:
        """
        获取末端执行器位姿
        
        Returns:
            pose: 末端位姿
        """
        pass
    
    @abstractmethod
    def enable(self) -> bool:
        """使能机器人"""
        pass
    
    @abstractmethod
    def disable(self) -> bool:
        """失能机器人"""
        pass
    
    @abstractmethod
    def emergency_stop(self) -> None:
        """急停"""
        pass
```

**实现示例 1**: MuJoCo 仿真控制器

```python
class MuJoCoController(RobotController):
    """MuJoCo 仿真控制器"""
    
    def __init__(self, mj_model, mj_data, arm_dof: int = 6):
        self.model = mj_model
        self.data = mj_data
        self.arm_dof = arm_dof
        self.enabled = True
    
    def connect(self) -> bool:
        return True  # 仿真无需连接
    
    def disconnect(self) -> None:
        pass
    
    def get_joint_state(self) -> JointState:
        return JointState(
            position=self.data.qpos[:self.arm_dof].copy(),
            velocity=self.data.qvel[:self.arm_dof].copy(),
            effort=self.data.qfrc_applied[:self.arm_dof].copy(),
            timestamp=self.data.time
        )
    
    def set_joint_command(self, mode: ControlMode, command: np.ndarray) -> bool:
        if not self.enabled:
            return False
        
        if mode == ControlMode.POSITION:
            self.data.ctrl[:self.arm_dof] = command
        elif mode == ControlMode.TORQUE:
            self.data.ctrl[:self.arm_dof] = command
        else:
            raise NotImplementedError(f"Mode {mode} not supported")
        
        return True
    
    def get_end_effector_pose(self) -> SE3:
        site_id = self.model.site("endpoint").id
        pos = self.data.site_xpos[site_id]
        mat = self.data.site_xmat[site_id].reshape(3, 3)
        return SE3(pos, mat)
```

### 2.3 Simulator 接口

**职责**: 仿真环境管理

```python
from typing import Dict, Any, List, Optional

@dataclass
class Observation:
    """观测数据"""
    joint_state: JointState
    end_effector_pose: SE3
    images: Dict[str, np.ndarray]  # camera_name -> image
    timestamp: float


class Simulator(ABC):
    """仿真器抽象接口"""
    
    @abstractmethod
    def reset(self, initial_state: Optional[np.ndarray] = None) -> Observation:
        """
        重置环境
        
        Args:
            initial_state: 初始关节状态 (可选)
        
        Returns:
            obs: 初始观测
        """
        pass
    
    @abstractmethod
    def step(self, action: np.ndarray) -> Observation:
        """
        执行一步仿真
        
        Args:
            action: 动作 (关节命令)
        
        Returns:
            obs: 观测
        """
        pass
    
    @abstractmethod
    def get_observation(self) -> Observation:
        """获取当前观测"""
        pass
    
    @abstractmethod
    def render(self, mode: str = "human") -> Optional[np.ndarray]:
        """
        渲染
        
        Args:
            mode: 'human' | 'rgb_array'
        
        Returns:
            image: 如果 mode='rgb_array', 返回图像
        """
        pass
    
    @abstractmethod
    def close(self) -> None:
        """关闭仿真器"""
        pass
```

### 5.3 错误处理示例

```python
class SafeIKController:
    """带安全检查的 IK 控制器"""
    
    def __init__(self, solver: IKSolver, controller: RobotController):
        self.solver = solver
        self.controller = controller
    
    def move_to_pose(self, target_pose: SE3) -> ErrorCode:
        try:
            # 1. 工作空间检查
            if not self._is_reachable(target_pose):
                return ErrorCode.IK_OUT_OF_WORKSPACE
            
            # 2. IK 求解
            current_state = self.controller.get_joint_state()
            q_target, converged = self.solver.solve(
                target_pose, 
                current_state.position, 
                dt=0.01
            )
            
            if not converged:
                return ErrorCode.IK_NOT_CONVERGED
            
            # 3. 关节限位检查
            if not self._check_joint_limits(q_target):
                return ErrorCode.JOINT_LIMIT_EXCEEDED
            
            # 4. 发送命令
            success = self.controller.set_joint_command(
                ControlMode.POSITION, 
                q_target
            )
            
            if not success:
                return ErrorCode.CONNECTION_LOST
            
            return ErrorCode.SUCCESS
        
        except Exception as e:
            print(f"Unexpected error: {e}")
            self.controller.emergency_stop()
            return ErrorCode.EMERGENCY_STOP
    
    def _is_reachable(self, pose: SE3) -> bool:
        """检查位姿是否在工作空间内"""
        distance = np.linalg.norm(pose.position)
        return 0.1 < distance < 0.6
    
    def _check_joint_limits(self, q: np.ndarray) -> bool:
        """检查关节是否在限位内"""
        return np.all(q >= self.q_min) and np.all(q <= self.q_max)
```

def check_compatibility(required_version: str, current_version: str) -> bool:
    """检查版本兼容性"""
    req_major, req_minor, _ = map(int, required_version.split('.'))
    cur_major, cur_minor, _ = map(int, current_version.split('.'))
    
    # 主版本号必须相同

        return False

        return False
    
    return True
