```python
def converge_ik(self, dt):
    print(f"[DEBUG] === IK Convergence Start ===")
    for i in range(self.MINK_MAX_ITERS):
        vel = mink.solve_ik(self.configuration, self.mink_tasks, dt, self.MINK_SOLVER, 1e-3)
        self.configuration.integrate_inplace(vel, dt)
        err = self.end_effector_task.compute_error(self.configuration)
        
        pos_err = np.linalg.norm(err[:3])
        ori_err = np.linalg.norm(err[3:])
        print(f"[DEBUG] Iter {i+1}: pos_err={pos_err*1000:.3f}mm, ori_err={ori_err:.5f}")
        
        pos_achieved = pos_err <= self.MINK_POS_THRESHOLD
        ori_achieved = ori_err <= self.MINK_ORI_THRESHOLD
        
        if pos_achieved and ori_achieved:
            print(f"[DEBUG] Converged in {i+1} iterations!")
            return True
    
    print(f"[DEBUG] Failed to converge after {self.MINK_MAX_ITERS} iterations")
    return False
```

```python
# 创建文件: examples/mocap_ik/profiler.py
import time
from contextlib import contextmanager
from collections import defaultdict
import numpy as np

class Profiler:
    def __init__(self):
        self.timings = defaultdict(list)
    
    @contextmanager
    def profile(self, name: str):
        start = time.perf_counter()
        try:
            yield
        finally:
            elapsed = (time.perf_counter() - start) * 1000  # ms
            self.timings[name].append(elapsed)
    
    def report(self):
        print("\n=== Performance Report ===")
        for name, times in self.timings.items():
            print(f"{name}:")
            print(f"  Mean: {np.mean(times):.2f}ms")
            print(f"  Std:  {np.std(times):.2f}ms")
            print(f"  Min:  {np.min(times):.2f}ms")
            print(f"  Max:  {np.max(times):.2f}ms")
            print(f"  P95:  {np.percentile(times, 95):.2f}ms")

# 全局 profiler
profiler = Profiler()
```

### 4.1 绘制 IK 收敛曲线

**创建可视化工具**:

```python
# 创建文件: examples/mocap_ik/visualizer.py
import matplotlib.pyplot as plt
import numpy as np
from collections import deque

class IKVisualizer:
    def __init__(self, max_history=1000):
        self.errors = deque(maxlen=max_history)
        self.iterations = deque(maxlen=max_history)
        self.convergence_flags = deque(maxlen=max_history)
        
        plt.ion()
        self.fig, self.axes = plt.subplots(2, 1, figsize=(10, 8))
    
    def add_result(self, error_history: list, converged: bool):
        """记录一次 IK 求解结果"""
        self.errors.append(error_history)
        self.iterations.append(len(error_history))
        self.convergence_flags.append(converged)
    
    def update_plot(self):
        """更新图表"""
        # 清空
        for ax in self.axes:
            ax.clear()
        
        # 图1: 最近一次收敛曲线
        if self.errors:
            last_errors = self.errors[-1]
            self.axes[0].plot(last_errors, 'o-')
            self.axes[0].axhline(y=1e-3, color='r', linestyle='--', label='Threshold')
            self.axes[0].set_xlabel('Iteration')
            self.axes[0].set_ylabel('Error (m)')
            self.axes[0].set_title('Last IK Convergence')
            self.axes[0].legend()
            self.axes[0].grid(True)
        
        # 图2: 迭代次数统计
        if self.iterations:
            self.axes[1].hist(list(self.iterations), bins=10, alpha=0.7)
            self.axes[1].set_xlabel('Iterations')
            self.axes[1].set_ylabel('Frequency')
            self.axes[1].set_title(f'Iteration Distribution (Conv Rate: {np.mean(self.convergence_flags)*100:.1f}%)')
            self.axes[1].grid(True)
        
        plt.tight_layout()
        plt.pause(0.01)

# 全局 visualizer
visualizer = IKVisualizer()
```

## 7. 进阶：简化版 IK 控制器

**目标**: 理解 IK 原理，实现一个最简单的雅可比逆解法

```python
# 创建文件: examples/mocap_ik/simple_ik.py
import numpy as np
import mujoco

class SimpleJacobianIK:
    """简化版雅可比逆解 IK (仅用于学习)"""
    
    def __init__(self, mj_model, mj_data, arm_dof=6):
        self.model = mj_model
        self.data = mj_data
        self.arm_dof = arm_dof
        
        # 获取末端 site ID
        self.site_id = mj_model.site("endpoint").id
    
    def solve_one_step(self, target_pos, dt=0.01):
        """
        单步雅可比逆解
        
        Returns:
            dq: 关节速度增量
        """
        # 1. 获取当前末端位置
        current_pos = self.data.site_xpos[self.site_id]
        
        # 2. 计算位置误差
        error = target_pos - current_pos
        v_desired = error / dt  # 期望速度
        
        # 3. 计算雅可比矩阵 (仅位置部分, 3×n)
        jacp = np.zeros((3, self.model.nv))
        mujoco.mj_jacSite(self.model, self.data, jacp, None, self.site_id)
        J = jacp[:, :self.arm_dof]
        
        # 4. 伪逆求解: dq = J^+ * v
        J_pinv = np.linalg.pinv(J)
        dq = J_pinv @ v_desired
        
        return dq
    
    def solve(self, target_pos, max_iters=10, threshold=1e-3):
        """
        迭代求解
        """
        for i in range(max_iters):
            # 单步求解
            dq = self.solve_one_step(target_pos, dt=0.01)
            
            # 积分更新
            self.data.qpos[:self.arm_dof] += dq * 0.01
            mujoco.mj_forward(self.model, self.data)
            
            # 检查误差
            current_pos = self.data.site_xpos[self.site_id]
            error = np.linalg.norm(target_pos - current_pos)
            
            print(f"Iter {i+1}: error = {error*1000:.3f}mm")
            
            if error < threshold:
                print(f"Converged in {i+1} iterations!")
                return True
        
        print("Failed to converge")
        return False


# 测试
```
